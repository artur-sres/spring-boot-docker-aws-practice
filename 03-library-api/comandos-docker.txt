# Checar porta
netstat -aof | findstr 5433

# Criar network
docker network create library-network

# Rodar container Postgre na versão 18.6
docker run -d --name librarydb --network library-network -p 5433:5432 -e POSTGRES_PASSWORD=postgres -e POSTGRES_USER=postgres -e POSTGRES_DB=library postgres:18.6

# Rodar Pgadmin4
docker run -d --name pgadmin4 --network library-network -p 15433:80 -e PGADMIN_DEFAULT_EMAIL=admin@admin.com -e PGADMIN_DEFAULT_PASSWORD=admin dpage/pgadmin4
